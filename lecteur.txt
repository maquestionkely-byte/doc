<script>
    // Configuration GitHub (à modifier selon votre dépôt)
    const REPO = 'maquestionkely-byte/doc';
    const BRANCH = 'main';
    
    // Récupérer le titre depuis l'URL
    const urlParams = new URLSearchParams(window.location.search);
    const titre = urlParams.get('titre');
    
    // Convertir le titre en nom de fichier
    function titreVersNomFichier(titre) {
        if (!titre) return null;
        return titre.toLowerCase().replace(/[^a-z0-9]/g, '_') + '.txt';
    }
    
    async function chargerContenu() {
        const titreEl = document.getElementById('titre');
        const infoEl = document.getElementById('info');
        const contenuEl = document.getElementById('contenu');
        
        if (!titre) {
            titreEl.innerHTML = '❌ Erreur';
            infoEl.innerHTML = 'Aucun titre spécifié';
            contenuEl.innerHTML = '<div class="error">❌ Paramètre "titre" manquant dans l\'URL<br><br>Exemple: lecteur.txt?titre=ity%20tsara</div>';
            return;
        }
        
        const nomFichier = titreVersNomFichier(titre);
        titreEl.innerHTML = `📖 ${titre}`;
        infoEl.innerHTML = `📄 Chargement de : ${nomFichier}`;
        
        try {
            const url = `https://raw.githubusercontent.com/${REPO}/${BRANCH}/${nomFichier}`;
            const response = await fetch(url, { cache: 'no-cache' });
            
            if (!response.ok) {
                throw new Error(`Fichier introuvable (HTTP ${response.status})`);
            }
            
            const contenu = await response.text();
            contenuEl.innerHTML = contenu.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
            infoEl.innerHTML = `✅ Fichier chargé avec succès - ${contenu.length} caractères`;
            
        } catch (error) {
            contenuEl.innerHTML = `<div class="error">❌ Erreur : ${error.message}<br><br>Vérifiez que le fichier "${nomFichier}" existe sur GitHub.</div>`;
            infoEl.innerHTML = `❌ Erreur de chargement`;
        }
    }
    
    chargerContenu();
</script>

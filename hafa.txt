    <button id="openHtml2">Ouvrir 3333333333333 (local)</button>
    <div id="status"></div>

    <script>
        // Fonction pour ouvrir HTML2 depuis le disque local
        document.getElementById('openHtml2').addEventListener('click', async () => {
            try {
                const fileInput = document.createElement('input');
                fileInput.type = 'file';
                fileInput.accept = '.html';
                fileInput.onchange = async (e) => {
                    const file = e.target.files[0];
                    if (file && file.name === 'HTML2.html') {
                        const content = await file.text();
                        const blob = new Blob([content], { type: 'text/html' });
                        const url = URL.createObjectURL(blob);
                        window.open(url, '_blank');
                    } else {
                        document.getElementById('status').innerHTML = '❌ Veuillez sélectionner HTML2.html';
                    }
                };
                fileInput.click();
            } catch (error) {
                document.getElementById('status').innerHTML = '❌ Erreur: ' + error.message;
            }
        });
    </script>

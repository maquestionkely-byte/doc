    // Fonction pour obtenir 60 questions aléatoires uniques
    function getRandomQuestions(n) {
        const shuffled = [...fullBank];
        for (let i = shuffled.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
        }
        while(shuffled.length < n) shuffled.push(...fullBank.slice(0, n - shuffled.length));
        return shuffled.slice(0, n);
    }

        if (questionNumber === 20) {

                motivationDiv.style.display = "block";
            }
            setTimeout(() => { if(motivationDiv) motivationDiv.style.display = "none"; }, 4000);
        } 
        else if (questionNumber === 40) {
            if (currentScoreValue >= 30) {
                motivationDiv.innerHTML = "🏆 CHAPEAU ! Vous maîtrisez vraiment bien ! Plus de 30 bonnes réponses sur 40. Persévérez jusqu'à la fin ! 🎯";
                motivationDiv.style.display = "block";
            } else {
                motivationDiv.innerHTML = "📖 Continuez vos efforts ! Vous avez moins de 30 bonnes réponses pour le moment. Relisez les règles des adjectifs possessifs et démonstratifs, vous allez y arriver ! 🌈";

            }
            setTimeout(() => { if(motivationDiv) motivationDiv.style.display = "none"; }, 4500);
        }

    // Afficher la traduction malagasy de la question courante (toggle)
    function toggleTranslation() {
        const existingPopup = document.getElementById('dynamicTranslationPopup');
        if (existingPopup) {
            existingPopup.remove();
            currentTranslationVisible = false;
            return;
        }
        const qData = questions[currentIdx];
        const mgText = qData.mg || "Tsy misy fandikana amin'izao (traduction non disponible)";
        const popupDiv = document.createElement('div');
        popupDiv.id = 'dynamicTranslationPopup';
        popupDiv.className = 'translation-popup';
        popupDiv.innerHTML = `🇲🇬 ${mgText}`;
        const headerDiv = document.querySelector('.question-header');
        if (headerDiv) headerDiv.insertAdjacentElement('afterend', popupDiv);
        currentTranslationVisible = true;
        // auto disparition au bout de 5 sec ou au prochain changement de question
        setTimeout(() => {
            const p = document.getElementById('dynamicTranslationPopup');
            if(p) p.remove();
            currentTranslationVisible = false;
        }, 5000);
    }
    
    // Nettoyer traduction si présente
    function clearTranslation() {
        const pop = document.getElementById('dynamicTranslationPopup');
        if(pop) pop.remove();
        currentTranslationVisible = false;
    }
    
    // Rendu question courante
    function renderCurrent() {
        if (quizFinished) return;
        clearTranslation();
        const q = questions[currentIdx];
        const alreadyAnswered = (userSelections[currentIdx] !== null);
        const letters = ['A', 'B', 'C', 'D'];
        
        let optsHtml = '';
        q.options.forEach((opt, idx) => {
            const letter = letters[idx];
            let selectedClass = (alreadyAnswered && userSelections[currentIdx] === letter) ? 'selected' : '';
            optsHtml += `

                    <span class="option-text">${opt}</span>

            `;
        });

                <div class="question-text">${currentIdx+1}. ${q.q}</div>

                div.addEventListener('click', (e) => {
                    if (isLocked[currentIdx] || quizFinished) return;
                    const chosenLetter = div.getAttribute('data-opt-letter');
                    if (chosenLetter) handleAnswer(chosenLetter);
                });
            });
        } else if (alreadyAnswered) {
            const optionDivs = document.querySelectorAll('.option-card');
            optionDivs.forEach(div => {
                div.style.cursor = "default";
                div.style.opacity = "0.7";
                div.style.pointerEvents = "none";
            });
        }
        
        qCounterSpan.innerText = `Question ${currentIdx+1} / ${TOTAL}`;
    }
    
    // Traitement réponse sans aperçu, mais avec gestion des paliers et score
    function handleAnswer(selectedLetter) {
        if (isLocked[currentIdx] || quizFinished) return;
        isLocked[currentIdx] = true;
        
        const q = questions[currentIdx];
        const isCorrect = (selectedLetter === q.correct);
        userSelections[currentIdx] = selectedLetter;
        if (isCorrect) {
            currentScore++;
            updateScoreUI();
        }
        
        // Mise à jour visuelle sélection
        renderCurrent();
        
        // Vérifier paliers 20 et 40 (on le fait après la réponse, en fonction du nombre de questions répondues)
        const answeredCount = userSelections.filter(a => a !== null).length;
        // answeredCount est le nombre de questions déjà traitées (y compris celle-ci)
        if (answeredCount === 20 || answeredCount === 40) {
            showPalierMotivation(answeredCount, currentScore);
        }
        
        // Passage automatique
        setTimeout(() => {
            if (currentIdx + 1 < TOTAL) {
                currentIdx++;
                renderCurrent();
            } else {
                finishQuizAndShowFullReport();
            }
        }, 380);
    }
    
    // Afficher résultat final complet

            <div style="text-align:center"><span class="final-score">${totalCorrect} / ${TOTAL} (${percent}%)</span></div>
            <p style="text-align:center; margin: 15px 0;">${finalMessage}</p>
            <div class="questions-list">
                <h3>📋 Liste complète des 60 questions avec vos réponses et les corrections :</h3>
        `;
        
        for (let i = 0; i < TOTAL; i++) {

            const lettersIndex = ['A','B','C','D'];
            const isOk = (userChoice === q.correct);
            const userText = userChoice ? q.options[lettersIndex.indexOf(userChoice)] : "(non répondue)";
            const correctAnswerText = q.options[lettersIndex.indexOf(q.correct)];
            const statusIcon = isOk ? "✅" : "❌";
            const borderClass = isOk ? "correct" : "wrong";
            listHtml += `
                <div class="question-summary ${borderClass}">

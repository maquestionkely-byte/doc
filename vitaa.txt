<script>
    // Configuration IndexedDB
    const DB_NAME = 'HtmlStorageApp';
    const STORE_NAME = 'htmlFiles';
    const DOCUMENT_ID = 'qcmHtmlContent';

    // L'URL CORRECTE que vous avez fournie
    const SOURCE_URL = 'https://raw.githubusercontent.com/maquestionkely-byte/doc/refs/heads/main/prom.txt';

    // Éléments DOM
    const downloadBtn = document.getElementById('downloadAndStoreBtn');
    const displayBtn = document.getElementById('displayStoredBtn');
    const contentArea = document.getElementById('contentArea');
    const statusSpan = document.getElementById('statusMsg');

    // --- 1. Fonctions IndexedDB ---
    async function openDB() {
        return new Promise((resolve, reject) => {
            const request = indexedDB.open(DB_NAME, 1);
            request.onupgradeneeded = (event) => {
                const db = event.target.result;
                if (!db.objectStoreNames.contains(STORE_NAME)) {
                    db.createObjectStore(STORE_NAME, { keyPath: 'id' });
                }
            };
            request.onsuccess = () => resolve(request.result);
            request.onerror = () => reject(request.error);
        });
    }

    async function saveHtmlToDB(htmlContent) {
        const db = await openDB();
        return new Promise((resolve, reject) => {
            const transaction = db.transaction([STORE_NAME], 'readwrite');
            const store = transaction.objectStore(STORE_NAME);
            const data = { 
                id: DOCUMENT_ID, 
                html: htmlContent, 
                timestamp: Date.now(),
                sourceUrl: SOURCE_URL
            };
            const request = store.put(data);
            request.onsuccess = () => resolve();
            request.onerror = () => reject(request.error);
            transaction.oncomplete = () => db.close();
        });
    }

    async function loadHtmlFromDB() {
        const db = await openDB();
        return new Promise((resolve, reject) => {
            const transaction = db.transaction([STORE_NAME], 'readonly');
            const store = transaction.objectStore(STORE_NAME);
            const request = store.get(DOCUMENT_ID);
            request.onsuccess = () => resolve(request.result ? request.result.html : null);
            request.onerror = () => reject(request.error);
            transaction.oncomplete = () => db.close();
        });
    }

    // --- 2. Télécharger depuis GitHub ---
    async function downloadHtmlFromUrl() {
        statusSpan.textContent = '⏳ Téléchargement en cours...';
        try {
            const response = await fetch(SOURCE_URL, { cache: 'no-store' });
            if (!response.ok) {
                throw new Error(`HTTP ${response.status}: ${response.statusText}`);
            }
            const htmlContent = await response.text();
            
            // Vérification minimale : s'assurer que c'est bien du HTML (ou au moins du texte)
            if (!htmlContent || htmlContent.trim().length < 20) {
                throw new Error('Le fichier téléchargé semble vide ou trop court.');
            }
            
            await saveHtmlToDB(htmlContent);
            statusSpan.textContent = '✅ HTML stocké dans IndexedDB !';
            return htmlContent;
        } catch (error) {
            console.error('Erreur téléchargement:', error);
            statusSpan.textContent = `❌ Échec téléchargement: ${error.message}`;
            throw error;
        }
    }

    // --- 3. Afficher le HTML (dans un iframe pour isolation et exécution correcte) ---
    function displayHtmlInIframe(htmlContent) {
        // Créer un iframe pour afficher le HTML de manière isolée,
        // ce qui permet d'exécuter son CSS et JavaScript sans conflit avec la page parente.
        const iframe = document.createElement('iframe');
        iframe.style.width = '100%';
        iframe.style.minHeight = '600px';
        iframe.style.border = 'none';
        iframe.style.backgroundColor = '#fff';
        
        // Nettoyer la zone de contenu
        contentArea.innerHTML = '';
        contentArea.appendChild(iframe);
        
        // Écrire le contenu HTML dans l'iframe
        const iframeDoc = iframe.contentWindow.document;
        iframeDoc.open();
        iframeDoc.write(htmlContent);
        iframeDoc.close();
        
        // Ajuster la hauteur de l'iframe en fonction du contenu (optionnel)
        iframe.onload = () => {
            try {
                const bodyHeight = iframe.contentWindow.document.body.scrollHeight;
                iframe.style.height = Math.max(600, bodyHeight + 20) + 'px';
            } catch(e) { /* sécurité cross-origin non appliquée ici car c'est du même domaine */ }
        };
    }

    // Afficher en mode texte brut (si échec de l'iframe)
    function displayAsPlainText(htmlContent) {
        contentArea.innerHTML = `<pre style="white-space: pre-wrap; word-wrap: break-word; background:#f4f4f4; padding:1rem; margin:1rem; border-radius:8px; max-height:600px; overflow:auto;">${escapeHtml(htmlContent)}</pre>`;
    }

    // Utilitaire pour échapper le HTML (affichage texte)
    function escapeHtml(str) {
        return str.replace(/[&<>]/g, function(m) {
            if (m === '&') return '&amp;';
            if (m === '<') return '&lt;';
            if (m === '>') return '&gt;';
            return m;
        }).replace(/[\uD800-\uDBFF][\uDC00-\uDFFF]/g, function(c) {
            return c;
        });
    }

    // --- 4. Afficher le contenu stocké (avec détection si c'est du HTML valide) ---
    async function displayStoredHtml() {
        statusSpan.textContent = '⏳ Lecture depuis IndexedDB...';
        try {
            const htmlContent = await loadHtmlFromDB();
            if (!htmlContent) {
                contentArea.innerHTML = `<div class="error-message">⚠️ Aucun HTML stocké. Veuillez d'abord cliquer sur "Télécharger & Stocker".</div>`;
                statusSpan.textContent = '⚠️ Rien en cache.';
                return;
            }
            
            // Tenter d'afficher dans un iframe
            // Vérification rapide : si le contenu semble être du HTML (contient <html> ou <body> ou <div> etc.)
            if (htmlContent.includes('<html') || htmlContent.includes('<!DOCTYPE') || htmlContent.includes('<body') || htmlContent.includes('<div')) {
                displayHtmlInIframe(htmlContent);
                statusSpan.textContent = '✅ HTML chargé depuis IndexedDB et affiché.';
            } else {
                // Si ce n'est pas du HTML, on l'affiche en texte
                displayAsPlainText(htmlContent);
                statusSpan.textContent = '📄 Contenu texte affiché (non HTML).';
            }
        } catch (error) {
            console.error(error);
            contentArea.innerHTML = `<div class="error-message">❌ Erreur lors de la lecture depuis IndexedDB : ${error.message}</div>`;
            statusSpan.textContent = '❌ Erreur lecture DB.';
        }
    }

    // --- 5. Télécharger ET ensuite afficher automatiquement ---
    async function downloadAndThenDisplay() {
        try {
            const htmlContent = await downloadHtmlFromUrl();
            // Option : afficher après téléchargement
            if (htmlContent) {
                displayHtmlInIframe(htmlContent);
                statusSpan.textContent = '✅ Téléchargé, stocké et affiché.';
            }
        } catch (error) {
            // En cas d'erreur, tenter d'afficher l'ancienne version en cache si elle existe
            const cachedHtml = await loadHtmlFromDB();
            if (cachedHtml) {
                displayHtmlInIframe(cachedHtml);
                statusSpan.textContent = '⚠️ Utilisation du cache (échec téléchargement).';
            } else {
                contentArea.innerHTML = `<div class="error-message">❌ Impossible de télécharger et aucun cache disponible. Vérifiez l'URL ou votre connexion.</div>`;
            }
        }
    }

    // --- Événements des boutons ---
    downloadBtn.addEventListener('click', downloadAndThenDisplay);
    displayBtn.addEventListener('click', displayStoredHtml);

    // Au chargement de la page, on vérifie si quelque chose est déjà stocké pour l'indiquer
    window.addEventListener('DOMContentLoaded', async () => {
        const existing = await loadHtmlFromDB();
        if (existing) {
            statusSpan.innerHTML = '📦 Un fichier HTML est déjà en cache. Cliquez sur "Afficher" ou "Mettre à jour".';
        }
    });
</script>

    function sauvegarderContexte(leconTitre, code4, niveau) {
        dernierContexte = {
            leconTitre: leconTitre,
            code4: code4,
            niveau: niveau,
            timestamp: Date.now()
        };
        localStorage.setItem('dernierContexteExercice', JSON.stringify(dernierContexte));
        localStorage.setItem('pagePrincipaleUrl', originalPageUrl);

    // Fonction pour revenir à la liste des exercices
    function revenirAListeExercices() {
        // Récupérer le contexte sauvegardé
        const contexteStr = localStorage.getItem('dernierContexteExercice');
        if (contexteStr) {
            const contexte = JSON.parse(contexteStr);
            if (contexte.leconTitre) {
                // Revenir à la page principale
                window.location.href = localStorage.getItem('pagePrincipaleUrl') || originalPageUrl;
                return;
            }
        }
        // Fallback: revenir simplement en arrière

            };
            
            request.onsuccess = function(event) {
                const db = event.target.result;
                const transaction = db.transaction([STORE_NAME], "readwrite");
                const store = transaction.objectStore(STORE_NAME);
                
                const putRequest = store.put(contenuHTML, "htmlContent");
                
                putRequest.onsuccess = function() {
                    console.log(`💾 Stocké dans ${DB_NAME}.${STORE_NAME}`);
                    db.close();
                    resolve();
                };
                
                putRequest.onerror = function(event) {
                    console.error("❌ Erreur stockage:", event.target.error);
                    db.close();
                    reject(event.target.error);
                };
            };
            
            request.onupgradeneeded = function(event) {
                const db = event.target.result;
                if (!db.objectStoreNames.contains(STORE_NAME)) {
                    db.createObjectStore(STORE_NAME);
                    console.log(`📁 Store créé: ${STORE_NAME}`);
                }
            };
        });
    }

    async function telechargerEtOuvrir(leconTitre, typeCode, niveau) {
        const code4 = get4Caracteres(leconTitre);
        const nomFichier = `${code4}${typeCode}.txt`;
        const url = `https://raw.githubusercontent.com/maquestionkely-byte/doc/refs/heads/main/${nomFichier}`;
        
        // Sauvegarder le contexte pour pouvoir revenir
        sauvegarderContexte(leconTitre, code4, niveau);
        
        const modal = document.createElement('div');
        modal.className = 'modal-fullscreen';
        modal.innerHTML = `
            <div class="modal-fullscreen-header">
                <h2><i class="fas fa-download"></i> Téléchargement - ${leconTitre}</h2>
                <button class="btn-close-fullscreen" id="closeLoadingBtn">✕</button>
            </div>
            <div class="modal-fullscreen-body" style="display: flex; align-items: center; justify-content: center;">
                <div style="text-align: center;">
                    <i class="fas fa-spinner fa-pulse" style="font-size: 3rem; color: #1d4e6e;"></i>
                    <p style="margin-top: 20px;">Téléchargement de ${nomFichier}...</p>
                    <p style="font-size: 0.8rem; color: #666; margin-top: 10px;">📀 DB: ${code4}${typeCode} (v7)</p>
                </div>
            </div>
        `;
        document.body.appendChild(modal);
        
        const closeBtn = modal.querySelector('#closeLoadingBtn');
        closeBtn.addEventListener('click', () => modal.remove());
        
        try {
            const response = await fetch(url);
            if (!response.ok) {
                throw new Error(`Fichier non trouvé (HTTP ${response.status})`);
            }
            const contenu = await response.text();
            
            await stockerEnIndexedDB(nomFichier, contenu);
            modal.remove();
            showToast(`✅ ${nomFichier} stocké dans IndexedDB`);
            
            // Créer le blob et naviguer
            const blob = new Blob([contenu], { type: 'text/html' });
            const blobUrl = URL.createObjectURL(blob);
            
            // Sauvegarder les infos de l'exercice pour la barre de navigation
            sessionStorage.setItem('exerciceInfo', JSON.stringify({
                leconTitre: leconTitre,
                typeCode: typeCode,
                typeNom: exercicesEcrits.find(e => e.code === typeCode)?.nom || typeCode
            }));
            
            window.location.href = blobUrl;
            
        } catch (error) {
            modal.remove();
            afficherErreur(url, nomFichier, code4, typeCode);
        }
    }

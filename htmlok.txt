    // ==================== CONFIGURATION ====================
    const TXT_URL = 'https://raw.githubusercontent.com/maquestionkely-byte/doc/main/Format.txt';
    const DATE_URL = 'https://raw.githubusercontent.com/maquestionkely-byte/doc/refs/heads/main/date_mise_a_jour.txt';
    const DB_NAME = 'QCM_Database';
    const DB_VERSION = 8;
    const STORE_NAME = 'qcm_data';
    const CHECK_INTERVAL_MS = 60000; // 1 minute = 60000 ms
    
    let db = null;
    let originalQuestions = [];
    let currentQuestions = [];
    let quizTitle = "QCM - Possessifs & Démonstratifs";
    let currentIdx = 0, userSelections = [], isLocked = [], currentScore = 0, quizFinished = false, isLoading = false;
    let lastKnownUpdateDate = null;  // Stocke la dernière date de mise à jour connue
    let lastRemoteDate = null;       // Stocke la date distante
    let checkInterval = null;
    let nextCheckTimeout = null;
    
    const contentArea = document.getElementById('contentArea');
    const quizTitleEl = document.getElementById('quizTitle');
    const updateBtn = document.getElementById('updateBtn');
    const cacheInfo = document.getElementById('cacheInfo');
    const updateModal = document.getElementById('updateModal');
    const confirmUpdateBtn = document.getElementById('confirmUpdateBtn');
    const cancelUpdateBtn = document.getElementById('cancelUpdateBtn');
    const modalDateInfo = document.getElementById('modalDateInfo');
    
    // ==================== MÉLANGE ALEATOIRE ====================
    function shuffleArray(array) {
        const shuffled = [...array];
        for (let i = shuffled.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
        }
        return shuffled;
    }
    
    function shuffleQuestions() {
        if (!originalQuestions || originalQuestions.length === 0) return;
        currentQuestions = shuffleArray(originalQuestions);
        resetQuizState();
        showToast("🃏 Ordre des questions mélangé !");
    }
    
    // ==================== INDEXEDDB ====================
    function openDB() {
        return new Promise((resolve, reject) => {
            const request = indexedDB.open(DB_NAME, DB_VERSION);
            request.onerror = () => reject(request.error);
            request.onsuccess = () => { db = request.result; resolve(db); };
            request.onupgradeneeded = (event) => {
                const db = event.target.result;
                if (!db.objectStoreNames.contains(STORE_NAME)) {
                    db.createObjectStore(STORE_NAME, { keyPath: 'id' });
                }
            };
        });
    }
    
    async function saveToDB(data) {
        return new Promise((resolve, reject) => {
            const transaction = db.transaction([STORE_NAME], 'readwrite');
            const store = transaction.objectStore(STORE_NAME);
            const request = store.put({ id: 'qcm_data', ...data, timestamp: Date.now() });
            request.onsuccess = () => resolve();
            request.onerror = () => reject(request.error);
        });
    }
    
    async function loadFromDB() {
        return new Promise((resolve, reject) => {
            const transaction = db.transaction([STORE_NAME], 'readonly');
            const store = transaction.objectStore(STORE_NAME);
            const request = store.get('qcm_data');
            request.onsuccess = () => resolve(request.result);
            request.onerror = () => reject(request.error);
        });
    }
    
    async function clearCache() {
        return new Promise((resolve, reject) => {
            const transaction = db.transaction([STORE_NAME], 'readwrite');
            const store = transaction.objectStore(STORE_NAME);
            const request = store.delete('qcm_data');
            request.onsuccess = () => resolve();
            request.onerror = () => reject(request.error);
        });
    }
    
    // ==================== VÉRIFICATION DE LA DATE DE MISE À JOUR ====================
    async function getRemoteUpdateDate() {
        try {
            const response = await fetch(DATE_URL, { cache: 'no-cache' });
            if (!response.ok) throw new Error(`HTTP ${response.status}`);
            const dateText = await response.text();
            const cleanDate = dateText.trim();
            console.log("Date distante reçue:", cleanDate);
            return cleanDate;
        } catch (error) {
            console.error("Erreur récupération date:", error);
            return null;
        }
    }
    
    // Sauvegarde la date de mise à jour dans IndexedDB
    async function saveUpdateDateToDB(date) {
        return new Promise((resolve, reject) => {
            const transaction = db.transaction([STORE_NAME], 'readwrite');
            const store = transaction.objectStore(STORE_NAME);
            const request = store.put({ id: 'update_date', date: date, timestamp: Date.now() });
            request.onsuccess = () => resolve();
            request.onerror = () => reject(request.error);
        });
    }
    
    async function loadUpdateDateFromDB() {
        return new Promise((resolve, reject) => {
            const transaction = db.transaction([STORE_NAME], 'readonly');
            const store = transaction.objectStore(STORE_NAME);
            const request = store.get('update_date');
            request.onsuccess = () => {
                const result = request.result;
                resolve(result ? result.date : null);
            };
            request.onerror = () => reject(request.error);
        });
    }
    
    // Mettre à jour l'affichage du prochain check
    function updateNextCheckDisplay() {
        const nextCheckSpan = document.getElementById('nextCheckSpan');
        if (nextCheckSpan && checkInterval) {
            const now = Date.now();
            const nextCheckTime = Math.ceil((CHECK_INTERVAL_MS - (now % CHECK_INTERVAL_MS)) / 1000);
            nextCheckSpan.textContent = `prochain check dans ${nextCheckTime}s`;
        }
    }
    
    // Vérification périodique de mise à jour
    async function checkForUpdate(isFirstCheck = false) {
        console.log(`🔍 Vérification de mise à jour ${isFirstCheck ? '(chargement page)' : '(périodique 1min)'}...`);
        
        const remoteDate = await getRemoteUpdateDate();
        if (!remoteDate) {
            console.log("Impossible de récupérer la date distante");
            updateNextCheckDisplay();
            return false;
        }
        
        lastRemoteDate = remoteDate;
        
        // Charger la dernière date connue depuis IndexedDB
        let storedDate = await loadUpdateDateFromDB();
        if (!storedDate && lastKnownUpdateDate) storedDate = lastKnownUpdateDate;
        
        console.log(`Comparaison: distante=${remoteDate}, stockée=${storedDate}`);
        
        if (storedDate !== remoteDate) {
            console.log("🆕 Nouvelle mise à jour détectée !");
            showUpdatePopup(remoteDate, storedDate);
            updateNextCheckDisplay();
            return true;
        }
        
        // Mettre à jour l'affichage avec l'heure de la dernière vérification
        const lastCheckTime = new Date().toLocaleTimeString();
        const statusSpan = document.getElementById('checkStatusSpan');
        if (statusSpan) {
            statusSpan.innerHTML = `✅ Dernier check: ${lastCheckTime}`;
        }
        
        updateNextCheckDisplay();
        return false;
    }
    
    // Afficher le popup de mise à jour
    function showUpdatePopup(newDate, oldDate) {
        if (updateModal.style.display === 'flex') return;
        
        if (oldDate) {
            modalDateInfo.innerHTML = `📅 Ancienne version: ${oldDate}<br>🆕 Nouvelle version: ${newDate}`;
        } else {
            modalDateInfo.innerHTML = `📅 Date de mise à jour: ${newDate}`;
        }
        
        updateModal.style.display = 'flex';
    }
    
    async function performUpdate() {
        updateModal.style.display = 'none';
        await forceUpdate();
    }
    
    // ==================== PARSER LE FICHIER TEXTE ====================
    function parseTextFile(content) {
        const lines = content.split('\n');
        let title = "QCM - Possessifs & Démonstratifs";
        const questionsList = [];
        let currentQuestion = null, currentOptions = [], currentCorrect = null;
        
        for (let i = 0; i < lines.length; i++) {
            let line = lines[i].trim();
            if (line === "") continue;
            if (line.startsWith('TITRE:')) {
                title = line.replace('TITRE:', '').trim();
                quizTitleEl.innerText = title;
                continue;
            }
            if (line.startsWith('## CORRECTION')) break;
            
            const questionMatch = line.match(/^(\d+)\.\s+(.+)$/);
            if (questionMatch) {
                if (currentQuestion !== null && currentOptions.length > 0) {
                    questionsList.push({ q: currentQuestion, options: [...currentOptions], correct: currentCorrect });
                }
                currentQuestion = questionMatch[2];
                currentOptions = [];
                currentCorrect = null;
                continue;
            }
            
            const starOptionMatch = line.match(/^\*([A-D])\)\s+(.+)$/);
            if (starOptionMatch && currentQuestion !== null) {
                currentCorrect = starOptionMatch[1];
                currentOptions.push(starOptionMatch[2].trim());
                continue;
            }
            
            const normalOptionMatch = line.match(/^([A-D])\)\s+(.+)$/);
            if (normalOptionMatch && currentQuestion !== null) {
                currentOptions.push(normalOptionMatch[2].trim());
                continue;
            }
        }
        
        if (currentQuestion !== null && currentOptions.length > 0) {
            questionsList.push({ q: currentQuestion, options: [...currentOptions], correct: currentCorrect });
        }
        
        console.log("✅ Questions parsées:", questionsList.length);
        return { title, questions: questionsList };
    }
    
    // ==================== TÉLÉCHARGEMENT ====================
    async function downloadFromServer() {
        try {
            // Télécharger le contenu du fichier Format.txt
            const response = await fetch(TXT_URL, { cache: 'no-cache' });
            if (response.status === 404) {
                throw new Error("Fichier introuvable. Vérifiez que le lien 'raw' GitHub est correct.");
            }
            if (!response.ok) throw new Error(`HTTP ${response.status}`);
            const content = await response.text();
            const parsed = parseTextFile(content);
            if (!parsed.questions || parsed.questions.length === 0) throw new Error("Aucune question trouvée");
            
            // Récupérer la date de mise à jour depuis le fichier distant
            const updateDate = await getRemoteUpdateDate();
            const finalUpdateDate = updateDate || new Date().toISOString();
            
            // Sauvegarder dans IndexedDB
            await saveToDB({ 
                title: parsed.title, 
                questions: parsed.questions, 
                lastUpdate: Date.now(),
                updateDate: finalUpdateDate
            });
            
            // Sauvegarder la date de mise à jour séparément
            await saveUpdateDateToDB(finalUpdateDate);
            lastKnownUpdateDate = finalUpdateDate;
            
            return { ...parsed, updateDate: finalUpdateDate };
        } catch (error) {
            console.error("Erreur téléchargement:", error);
            throw error;
        }
    }
    
    // ==================== MISE À JOUR FORCÉE ====================
    async function forceUpdate() {
        if (isLoading) { showToast("⏳ Déjà en cours..."); return; }
        isLoading = true;
        updateBtn.disabled = true;
        updateBtn.textContent = "⏳ Téléchargement...";
        contentArea.innerHTML = '<div class="loading">🔄 Mise à jour en cours...</div>';
        
        try {
            await clearCache();
            const data = await downloadFromServer();
            originalQuestions = data.questions;
            currentQuestions = shuffleArray(originalQuestions);
            quizTitle = data.title;
            quizTitleEl.innerText = quizTitle;
            resetQuizState();
            showToast("✅ Mise à jour réussie !");
            
            // Afficher la date de mise à jour
            const displayDate = data.updateDate || new Date().toLocaleString();
            updateCacheInfo(displayDate);
        } catch (error) {
            const cached = await loadFromDB();
            if (cached && cached.questions && cached.questions.length > 0) {
                originalQuestions = cached.questions;
                currentQuestions = shuffleArray(originalQuestions);
                quizTitle = cached.title;
                quizTitleEl.innerText = quizTitle;
                resetQuizState();
                showToast("⚠️ Utilisation du cache");
                const dateStr = cached.updateDate || new Date(cached.timestamp).toLocaleString();
                updateCacheInfo(dateStr, true);
            } else {
                contentArea.innerHTML = `<div class="error">❌ Erreur : ${error.message}<br><button onclick="location.reload()" style="margin-top:10px;padding:8px 16px;background:#1d4e6e;color:white;border:none;border-radius:40px;">🔄 Réessayer</button></div>`;
                showToast("❌ Échec de la mise à jour");
            }
        } finally {
            isLoading = false;
            updateBtn.disabled = false;
            updateBtn.textContent = "🔄 Mettre à jour";
        }
    }
    
    function updateCacheInfo(dateStr, isCache = false) {
        const prefix = isCache ? "⚠️ Cache" : "📅 Dernière mise à jour";
        cacheInfo.innerHTML = `${prefix} : ${dateStr}<br>
            <span id="checkStatusSpan">🔄 Vérification auto toutes les 1 min</span>
            <span id="nextCheckSpan" class="next-check"></span>`;
    }
    
    // ==================== CHARGEMENT ====================
    async function loadQCMData() {
        try {
            await openDB();
            
            // Charger la dernière date de mise à jour connue
            lastKnownUpdateDate = await loadUpdateDateFromDB();
            
            const cached = await loadFromDB();
            if (cached && cached.questions && cached.questions.length > 0) {
                const dateStr = cached.updateDate || new Date(cached.timestamp).toLocaleString();
                updateCacheInfo(dateStr, true);
                return { title: cached.title, questions: cached.questions };
            }
            updateCacheInfo("Premier téléchargement...");
            return await downloadFromServer();
        } catch (error) {
            const cached = await loadFromDB();
            if (cached && cached.questions && cached.questions.length > 0) {
                updateCacheInfo("Cache de secours", true);
                return { title: cached.title, questions: cached.questions };
            }
            throw error;
        }
    }
    
    // ==================== QUIZ ====================
    function resetQuizState() {
        currentIdx = 0;
        userSelections = new Array(currentQuestions.length).fill(null);
        isLocked = new Array(currentQuestions.length).fill(false);
        currentScore = 0;
        quizFinished = false;
        renderCurrent();
    }
    
    function renderCurrent() {
        if (quizFinished || !currentQuestions || currentQuestions.length === 0) return;
        const q = currentQuestions[currentIdx];
        if (!q) return;
        
        const alreadyAnswered = (userSelections[currentIdx] !== null);
        const letters = ['A', 'B', 'C', 'D'];
        let optsHtml = '';
        q.options.forEach((opt, idx) => {
            const letter = letters[idx];
            const selectedClass = (alreadyAnswered && userSelections[currentIdx] === letter) ? 'selected' : '';
            optsHtml += `<div class="option-card ${selectedClass}" data-opt-letter="${letter}"><div class="option-letter">${letter}</div><span class="option-text">${opt}</span></div>`;
        });
        
        const html = `<div class="progress-bar"><span class="badge" id="qCounter">Question ${currentIdx+1} / ${currentQuestions.length}</span><span class="badge score-badge" id="liveScore">🎯 Score : ${currentScore} / ${currentQuestions.length}</span></div><div class="question-area"><div class="question-text">${currentIdx+1}. ${q.q}</div><div class="options-list" id="optionsContainer">${optsHtml}</div><button class="shuffle-btn" id="shuffleBtn">🃏 Mélanger l'ordre</button></div>`;
        contentArea.innerHTML = html;
        
        document.getElementById('shuffleBtn')?.addEventListener('click', () => {
            if (!quizFinished && !isLocked[currentIdx]) {
                shuffleQuestions();
            } else {
                showToast("❌ Terminez ou recommencez d'abord pour mélanger");
            }
        });
        
        if (!alreadyAnswered && !isLocked[currentIdx] && !quizFinished) {
            document.querySelectorAll('.option-card').forEach(div => {
                div.addEventListener('click', () => {
                    if (isLocked[currentIdx] || quizFinished) return;
                    const chosenLetter = div.getAttribute('data-opt-letter');
                    if (chosenLetter) handleAnswer(chosenLetter);
                });
            });
        }
    }
    
    function handleAnswer(selectedLetter) {
        if (isLocked[currentIdx] || quizFinished) return;
        isLocked[currentIdx] = true;
        const q = currentQuestions[currentIdx];
        if (selectedLetter === q.correct) currentScore++;
        userSelections[currentIdx] = selectedLetter;
        renderCurrent();
        setTimeout(() => {
            if (currentIdx + 1 < currentQuestions.length) { currentIdx++; renderCurrent(); }
            else finishQuiz();
        }, 300);
    }
    
    function finishQuiz() {
        quizFinished = true;
        const totalCorrect = currentScore;
        const percent = Math.round((totalCorrect / currentQuestions.length) * 100);
        let listHtml = `<div class="final-report"><div class="final-header">📊 RÉSULTAT FINAL</div><div class="text-center"><span class="final-score">${totalCorrect} / ${currentQuestions.length} (${percent}%)</span></div><div class="questions-list"><h3>📋 Corrigé complet :</h3>`;
        for (let i = 0; i < currentQuestions.length; i++) {
            const q = currentQuestions[i];
            const userChoice = userSelections[i];
            const lettersIndex = ['A','B','C','D'];
            const isOk = (userChoice === q.correct);
            const userText = userChoice ? q.options[lettersIndex.indexOf(userChoice)] : "(non répondue)";
            const correctAnswerText = q.options[lettersIndex.indexOf(q.correct)];
            listHtml += `<div class="question-summary ${isOk ? 'correct' : 'wrong'}"><strong>${isOk ? "✅" : "❌"} Q${i+1} : ${q.q}</strong><br>✏️ Votre réponse : ${userChoice || "—"} → "${userText}"<br>📌 Bonne réponse : ${q.correct} → "${correctAnswerText}"</div>`;
        }
        listHtml += `<button class="restart-btn" id="restartQuizBtn">🔄 Recommencer (même ordre)</button><button class="shuffle-btn" id="shuffleRestartBtn">🃏 Recommencer (ordre mélangé)</button></div></div>`;
        contentArea.innerHTML = `<div class="progress-bar"><span class="badge">Terminé !</span><span class="badge score-badge">🎯 Score final : ${totalCorrect} / ${currentQuestions.length}</span></div>${listHtml}`;
        
        document.getElementById('restartQuizBtn')?.addEventListener('click', () => resetQuizState());
        document.getElementById('shuffleRestartBtn')?.addEventListener('click', () => shuffleQuestions());
    }
    
    function showToast(message) {
        document.querySelector('.toast')?.remove();
        const toast = document.createElement('div');
        toast.className = 'toast';
        toast.textContent = message;
        document.body.appendChild(toast);
        setTimeout(() => toast.remove(), 3000);
    }
    
    // Mise à jour du compte à rebours
    function startCountdownUpdater() {
        setInterval(() => {
            updateNextCheckDisplay();
        }, 1000);
    }
    
    // Démarrer la vérification périodique (toutes les 1 minute)
    function startPeriodicCheck() {
        if (checkInterval) clearInterval(checkInterval);
        
        // Vérification immédiate au chargement
        setTimeout(() => {
            checkForUpdate(true);
        }, 1000);
        
        // Puis toutes les 1 minute
        checkInterval = setInterval(() => {
            checkForUpdate(false);
        }, CHECK_INTERVAL_MS);
        
        console.log("⏰ Vérification automatique démarrée (toutes les 1 minute)");
    }
    
    async function init() {
        try {
            const data = await loadQCMData();
            originalQuestions = data.questions;
            currentQuestions = shuffleArray(originalQuestions);
            quizTitle = data.title;
            quizTitleEl.innerText = quizTitle;
            if (!currentQuestions || currentQuestions.length === 0) throw new Error("Aucune question");
            resetQuizState();
            
            // Démarrer la vérification périodique
            startPeriodicCheck();
            startCountdownUpdater();
            
        } catch (error) {
            contentArea.innerHTML = `<div class="error">❌ Erreur : ${error.message}<br><button onclick="location.reload()" style="margin-top:10px;padding:8px 16px;background:#1d4e6e;color:white;border:none;border-radius:40px;">🔄 Réessayer</button></div>`;
        }
    }
